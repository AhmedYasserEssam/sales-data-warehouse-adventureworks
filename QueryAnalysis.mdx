-- 1) Total sales measures by order year
SELECT
{
    [Measures].[Gross Sales Amount],
    [Measures].[Discount Amount],
    [Measures].[Net Sales Amount],
    [Measures].[Margin Amount]
} ON COLUMNS,
NON EMPTY
[Order Date].[Year Num].[Year Num].Members ON ROWS
FROM [AdventureWorksDW_View];
-- 2) Net sales by order quarter
SELECT
{ [Measures].[Net Sales Amount] } ON COLUMNS,
NON EMPTY
[Order Date].[Year Quarter Name].[Year Quarter Name].Members ON ROWS
FROM [AdventureWorksDW_View];

-- 4) Sales by product category and year
SELECT
[Order Date].[Year Num].[Year Num].Members ON COLUMNS,
NON EMPTY
[DIM PRODUCT].[Category Name].[Category Name].Members ON ROWS
FROM [AdventureWorksDW_View]
WHERE ([Measures].[Net Sales Amount]);
-- 5) Top 10 products by margin
SELECT
{ [Measures].[Margin Amount] } ON COLUMNS,
TOPCOUNT(
    NONEMPTY(
        [DIM PRODUCT].[Product Name].[Product Name].Members,
        [Measures].[Margin Amount]
    ),
    10,
    [Measures].[Margin Amount]
) ON ROWS
FROM [AdventureWorksDW_View];
